cmake_minimum_required(VERSION 3.20)
project(Afterimage VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets OpenGL Network Multimedia)
find_package(OpenCV REQUIRED COMPONENTS core imgproc videoio)
find_package(Threads REQUIRED)

add_library(frame_engine src/engine.cpp)
target_include_directories(frame_engine PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(frame_engine PUBLIC ${OpenCV_LIBS})
add_library(gpu_engine src/gpu_engine.cpp)
target_link_libraries(gpu_engine PUBLIC frame_engine Qt6::OpenGL)
add_library(modulation src/modulation.cpp)
target_link_libraries(modulation PUBLIC frame_engine Qt6::Widgets Qt6::Network Qt6::Multimedia)

option(AFTERIMAGE_ENABLE_SYPHON "Enable macOS Syphon output" OFF)
if(APPLE AND AFTERIMAGE_ENABLE_SYPHON)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  find_library(SYPHON_FRAMEWORK Syphon HINTS "${SYPHON_ROOT}" REQUIRED)
  add_library(syphon_output src/syphon_output.mm)
  target_compile_options(syphon_output PRIVATE -fobjc-arc)
  target_link_libraries(syphon_output PUBLIC Qt6::Core "${SYPHON_FRAMEWORK}" "-framework OpenGL" "-framework Foundation")
else()
  add_library(syphon_output src/syphon_output_stub.cpp)
  target_link_libraries(syphon_output PUBLIC Qt6::Core)
endif()
target_include_directories(syphon_output PUBLIC src)

add_executable(afterimage MACOSX_BUNDLE src/main.cpp)
target_link_libraries(afterimage PRIVATE gpu_engine modulation syphon_output Qt6::Widgets Threads::Threads)
set_target_properties(afterimage PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "studio.afterimage.app")
if(APPLE)
  set_target_properties(afterimage PROPERTIES OUTPUT_NAME "Afterimage"
    MACOSX_BUNDLE_BUNDLE_NAME "Afterimage" MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/src/Info.plist.in")
endif()
install(TARGETS afterimage BUNDLE DESTINATION . RUNTIME DESTINATION bin)
if(APPLE)
  install(FILES docs/THIRD_PARTY_NOTICES.txt DESTINATION Afterimage.app/Contents/Resources)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(engine-test tests/engine_test.cpp)
  target_link_libraries(engine-test PRIVATE frame_engine)
  add_test(NAME engine COMMAND engine-test)
  add_executable(integration-test tests/integration_test.cpp)
  target_link_libraries(integration-test PRIVATE gpu_engine modulation syphon_output Qt6::Widgets Threads::Threads)
  add_test(NAME integration COMMAND integration-test)
  add_test(NAME gui-smoke COMMAND afterimage --smoke-test)
  set_tests_properties(integration gui-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=;QT_STYLE_OVERRIDE=Fusion" TIMEOUT 30)
  add_executable(modulation-test tests/modulation_test.cpp)
  target_link_libraries(modulation-test PRIVATE modulation)
  add_test(NAME modulation COMMAND modulation-test)
  set_tests_properties(modulation PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=;QT_STYLE_OVERRIDE=Fusion" TIMEOUT 30)
  add_executable(gpu-test tests/gpu_test.cpp)
  target_link_libraries(gpu-test PRIVATE gpu_engine)
  add_test(NAME gpu COMMAND gpu-test)
  set_tests_properties(gpu PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
  if(APPLE)
    set_tests_properties(gpu PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=cocoa;QT_QPA_PLATFORMTHEME=;AFTERIMAGE_REQUIRE_GPU=1")
  endif()
  if(APPLE AND AFTERIMAGE_ENABLE_SYPHON)
    add_executable(syphon-test tests/syphon_test.mm)
    target_compile_options(syphon-test PRIVATE -fobjc-arc)
    target_link_libraries(syphon-test PRIVATE gpu_engine syphon_output)
    add_test(NAME syphon COMMAND syphon-test)
    set_tests_properties(syphon PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=cocoa;QT_QPA_PLATFORMTHEME=" TIMEOUT 30)
  endif()
  add_test(NAME macos-plist COMMAND ${CMAKE_COMMAND}
    -DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR} -DTEST_BINARY_DIR=${CMAKE_CURRENT_BINARY_DIR}
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/macos_plist_test.cmake)
endif()

option(AFTERIMAGE_DEPLOY_MACOS "Bundle runtime dependencies when installing on macOS" OFF)
if(APPLE AND AFTERIMAGE_DEPLOY_MACOS)
  get_target_property(qmake_path Qt6::qmake IMPORTED_LOCATION)
  get_filename_component(qt_bin_dir "${qmake_path}" DIRECTORY)
  find_program(AFTERIMAGE_MACDEPLOYQT macdeployqt HINTS "${qt_bin_dir}" REQUIRED)
  set(AFTERIMAGE_LIBRARY_DIRS "${OpenCV_INSTALL_PATH}/lib;${qt_bin_dir}/../lib;${SYPHON_ROOT}")
  configure_file(cmake/DeployMac.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/DeployMac.cmake" @ONLY)
  install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/DeployMac.cmake")
endif()
